import React, { useState, useMemo } from 'react';

import { 

  Plus, 

  Search, 

  Filter, 

  Building2, 

  CheckCircle2, 

  PauseCircle, 

  AlertCircle,

  MoreVertical,

  Edit3,

  Trash2,

  Eye,

  EyeOff

} from 'lucide-react';

import { ModuleLayout } from '@/components/layout/ModuleLayout';

import { SplitViewLayout } from '@/components/layout/SplitViewLayout';

import { ConfirmModal } from '@/components/common/ConfirmModal';

import { PropertyCreatorWizard } from './PropertyCreatorWizard';

import { PropertyInlineEditor } from './PropertyInlineEditor';

import { useProperties } from '@/hooks/useProperties';

import { useToast } from '@/hooks/useToast';

import { Property, PropertyStatus } from '@/types/property';

export const AsesorInventoryView: React.FC = () => {

  const { showToast } = useToast();

  

  // React Query Integration

  const { 

    properties = [], 

    isLoading, 

    isError, 

    error,

    updatePropertyStatus, 

    deleteProperty 

  } = useProperties();

  // Local UI State

  const [searchTerm, setSearchTerm] = useState('');

  const [statusFilter, setStatusFilter] = useState<string>('ALL');

  const [selectedPropertyId, setSelectedPropertyId] = useState<string | null>(null);

  const [isCreatorOpen, setIsCreatorOpen] = useState(false);

  const [isEditorOpen, setIsEditorOpen] = useState(false);

  // Modal Confirmation States

  const [confirmModal, setConfirmModal] = useState<{

    isOpen: boolean;

    title: string;

    message: string;

    action: () => Promise<void>;

    variant?: 'danger' | 'warning' | 'info';

  }>({

    isOpen: false,

    title: '',

    message: '',

    action: async () => {},

    variant: 'info'

  });

  // Filtered Properties Computation

  const filteredProperties = useMemo(() => {

    return properties.filter((property: Property) => {

      const matchesSearch = 

        property.title.toLowerCase().includes(searchTerm.toLowerCase()) ||

        property.code.toLowerCase().includes(searchTerm.toLowerCase()) ||

        property.address.toLowerCase().includes(searchTerm.toLowerCase());

      

      const matchesStatus = statusFilter === 'ALL' || property.status === statusFilter;

      return matchesSearch && matchesStatus;

    });

  }, [properties, searchTerm, statusFilter]);

  // KPI Metrics Calculation

  const kpis = useMemo(() => {

    const total = properties.length;

    const active = properties.filter((p: Property) => p.status === 'PUBLISHED').length;

    const paused = properties.filter((p: Property) => p.status === 'PAUSED').length;

    const draft = properties.filter((p: Property) => p.status === 'DRAFT').length;

    return { total, active, paused, draft };

  }, [properties]);

  // Handlers for Status Changes with Concurrency Error Handling

  const handleToggleStatus = (property: Property) => {

    const isPublished = property.status === 'PUBLISHED';

    const newStatus: PropertyStatus = isPublished ? 'PAUSED' : 'PUBLISHED';

    const actionLabel = isPublished ? 'pausar' : 'publicar';

    setConfirmModal({

      isOpen: true,

      title: `${isPublished ? 'Pausar' : 'Publicar'} Propiedad`,

      message: `¿Estás seguro de que deseas ${actionLabel} la propiedad "${property.title}"?`,

      variant: isPublished ? 'warning' : 'info',

      action: async () => {

        try {

          await updatePropertyStatus.mutateAsync({

            id: property.id,

            status: newStatus,

            version: property.version

          });

          showToast({

            title: 'Estado actualizado',

            message: `La propiedad ahora está ${newStatus === 'PUBLISHED' ? 'publicada' : 'pausada'}.`,

            type: 'success'

          });

        } catch (err: any) {

          if (err?.code === '412_VERSION_OBSOLETA' || err?.status === 412) {

            showToast({

              title: 'Error de Concurrencia (412)',

              message: 'La propiedad fue modificada por otro usuario. La lista se ha actualizado con la versión más reciente.',

              type: 'error'

            });

          } else {

            showToast({

              title: 'Error al actualizar',

              message: err?.message || 'No se pudo cambiar el estado de la propiedad.',

              type: 'error'

            });

          }

        } finally {

          setConfirmModal((prev) => ({ ...prev, isOpen: false }));

        }

      }

    });

  };

  const handleDeleteProperty = (property: Property) => {

    setConfirmModal({

      isOpen: true,

      title: 'Eliminar Propiedad',

      message: `¿Estás seguro de que deseas eliminar permanentemente "${property.title}"? Esta acción no se puede deshacer.`,

      variant: 'danger',

      action: async () => {

        try {

          await deleteProperty.mutateAsync(property.id);

          showToast({

            title: 'Propiedad eliminada',

            message: 'La propiedad se ha removido del inventario correctamente.',

            type: 'success'

          });

        } catch (err: any) {

          showToast({

            title: 'Error al eliminar',

            message: err?.message || 'Ocurrió un error al intentar eliminar la propiedad.',

            type: 'error'

          });

        } finally {

          setConfirmModal((prev) => ({ ...prev, isOpen: false }));

        }

      }

    });

  };

  const handleOpenEditor = (propertyId: string) => {

    setSelectedPropertyId(propertyId);

    setIsEditorOpen(true);

    setIsCreatorOpen(false);

  };

  const handleOpenCreator = () => {

    setIsCreatorOpen(true);

    setIsEditorOpen(false);

    setSelectedPropertyId(null);

  };

  return (

    <ModuleLayout

      title="Inventario de Propiedades"

      subtitle="Gestiona la publicación, edición y estado de tus inmuebles."

      actions={

        <button

          onClick={handleOpenCreator}

          className="flex items-center gap-2 px-4 py-2 bg-blue-600 hover:bg-blue-700 text-white rounded-lg font-medium text-sm transition-colors"

        >

          <Plus className="w-4 h-4" />

          Nueva Propiedad

        </button>

      }

    >

      <SplitViewLayout

        isOpen={isCreatorOpen || isEditorOpen}

        onClose={() => {

          setIsCreatorOpen(false);

          setIsEditorOpen(false);

          setSelectedPropertyId(null);

        }}

        panel={

          isCreatorOpen ? (

            <PropertyCreatorWizard

              onClose={() => setIsCreatorOpen(false)}

            />

          ) : isEditorOpen && selectedPropertyId ? (

            <PropertyInlineEditor

              propertyId={selectedPropertyId}

              onClose={() => {

                setIsEditorOpen(false);

                setSelectedPropertyId(null);

              }}

            />

          ) : null

        }

      >

        <div className="grid grid-cols-1 lg:grid-cols-4 gap-6">

          {/* Panel de KPIs (Izquierda en Desktop) */}

          <div className="lg:col-span-1 space-y-4">

            <div className="bg-white p-4 rounded-xl border border-slate-200 shadow-sm">

              <div className="flex items-center gap-3">

                <div className="p-2 bg-slate-100 rounded-lg text-slate-600">

                  <Building2 className="w-5 h-5" />

                </div>

                <div>

                  <p className="text-xs text-slate-500 font-medium">Total Inmuebles</p>

                  <p className="text-xl font-bold text-slate-900">{kpis.total}</p>

                </div>

              </div>

            </div>

            <div className="bg-white p-4 rounded-xl border border-slate-200 shadow-sm">

              <div className="flex items-center gap-3">

                <div className="p-2 bg-emerald-50 rounded-lg text-emerald-600">

                  <CheckCircle2 className="w-5 h-5" />

                </div>

                <div>

                  <p className="text-xs text-slate-500 font-medium">Publicadas</p>

                  <p className="text-xl font-bold text-slate-900">{kpis.active}</p>

                </div>

              </div>

            </div>

            <div className="bg-white p-4 rounded-xl border border-slate-200 shadow-sm">

              <div className="flex items-center gap-3">

                <div className="p-2 bg-amber-50 rounded-lg text-amber-600">

                  <PauseCircle className="w-5 h-5" />

                </div>

                <div>

                  <p className="text-xs text-slate-500 font-medium">Pausadas</p>

                  <p className="text-xl font-bold text-slate-900">{kpis.paused}</p>

                </div>

              </div>

            </div>

            <div className="bg-white p-4 rounded-xl border border-slate-200 shadow-sm">

              <div className="flex items-center gap-3">

                <div className="p-2 bg-slate-50 rounded-lg text-slate-500">

                  <AlertCircle className="w-5 h-5" />

                </div>

                <div>

                  <p className="text-xs text-slate-500 font-medium">Borradores</p>

                  <p className="text-xl font-bold text-slate-900">{kpis.draft}</p>

                </div>

              </div>

            </div>

          </div>

          {/* Listado y Filtros (Centro/Derecha) */}

          <div className="lg:col-span-3 space-y-4">

            {/* Barra de Búsqueda y Filtro */}

            <div className="flex flex-col sm:flex-row gap-3 bg-white p-4 rounded-xl border border-slate-200 shadow-sm">

              <div className="relative flex-1">

                <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />

                <input

                  type="text"

                  placeholder="Buscar por título, código o dirección..."

                  value={searchTerm}

                  onChange={(e) => setSearchTerm(e.target.value)}

                  className="w-full pl-9 pr-4 py-2 border border-slate-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-blue-500/20 focus:border-blue-500"

                />

              </div>

              <div className="flex items-center gap-2">

                <Filter className="w-4 h-4 text-slate-400" />

                <select

                  value={statusFilter}

                  onChange={(e) => setStatusFilter(e.target.value)}

                  className="px-3 py-2 border border-slate-200 rounded-lg text-sm bg-white focus:outline-none focus:ring-2 focus:ring-blue-500/20 focus:border-blue-500"

                >

                  <option value="ALL">Todos los Estados</option>

                  <option value="PUBLISHED">Publicadas</option>

                  <option value="PAUSED">Pausadas</option>

                  <option value="DRAFT">Borradores</option>

                </select>

              </div>

            </div>

            {/* Estado de Carga y Errores */}

            {isLoading && (

              <div className="bg-white p-12 text-center rounded-xl border border-slate-200">

                <div className="inline-block animate-spin rounded-full h-8 w-8 border-4 border-slate-200 border-t-blue-600 mb-2"></div>

                <p className="text-sm text-slate-500">Cargando inventario...</p>

              </div>

            )}

            {isError && (

              <div className="bg-red-50 border border-red-200 rounded-xl p-4 text-red-700 text-sm">

                Ocurrió un error al cargar las propiedades: {error?.message || 'Error desconocido'}

              </div>

            )}

            {/* Tabla de Propiedades */}

            {!isLoading && !isError && (

              <div className="bg-white rounded-xl border border-slate-200 shadow-sm overflow-hidden">

                {filteredProperties.length === 0 ? (

                  <div className="p-12 text-center text-slate-500">

                    <Building2 className="w-12 h-12 text-slate-300 mx-auto mb-3" />

                    <p className="font-medium">No se encontraron propiedades</p>

                    <p className="text-xs text-slate-400 mt-1">Prueba cambiando los filtros o la búsqueda.</p>

                  </div>

                ) : (

                  <div className="overflow-x-auto">

                    <table className="w-full text-left text-sm">

                      <thead className="bg-slate-50 border-b border-slate-200 text-slate-600 text-xs uppercase font-semibold">

                        <tr>

                          <th className="px-4 py-3">Propiedad</th>

                          <th className="px-4 py-3">Código</th>

                          <th className="px-4 py-3">Precio</th>

                          <th className="px-4 py-3">Estado</th>

                          <th className="px-4 py-3 text-right">Acciones</th>

                        </tr>

                      </thead>

                      <tbody className="divide-y divide-slate-100">

                        {filteredProperties.map((property: Property) => (

                          <tr key={property.id} className="hover:bg-slate-50/80 transition-colors">

                            <td className="px-4 py-3">

                              <div className="font-medium text-slate-900">{property.title}</div>

                              <div className="text-xs text-slate-500 truncate max-w-xs">{property.address}</div>

                            </td>

                            <td className="px-4 py-3 font-mono text-xs text-slate-600">

                              {property.code}

                            </td>

                            <td className="px-4 py-3 font-semibold text-slate-900">

                              ${property.price?.toLocaleString()} {property.currency}

                            </td>

                            <td className="px-4 py-3">

                              <span className={`inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-full text-xs font-medium ${

                                property.status === 'PUBLISHED' 

                                  ? 'bg-emerald-50 text-emerald-700 border border-emerald-200'

                                  : property.status === 'PAUSED'

                                  ? 'bg-amber-50 text-amber-700 border border-amber-200'

                                  : 'bg-slate-100 text-slate-600 border border-slate-200'

                              }`}>

                                {property.status === 'PUBLISHED' && <CheckCircle2 className="w-3 h-3" />}

                                {property.status === 'PAUSED' && <PauseCircle className="w-3 h-3" />}

                                {property.status === 'DRAFT' && <AlertCircle className="w-3 h-3" />}

                                {property.status}

                              </span>

                            </td>

                            <td className="px-4 py-3 text-right">

                              <div className="flex items-center justify-end gap-1">

                                <button

                                  onClick={() => handleToggleStatus(property)}

                                  title={property.status === 'PUBLISHED' ? 'Pausar' : 'Publicar'}

                                  className="p-1.5 hover:bg-slate-100 rounded-lg text-slate-500 hover:text-slate-800 transition-colors"

                                >

                                  {property.status === 'PUBLISHED' ? (

                                    <EyeOff className="w-4 h-4" />

                                  ) : (

                                    <Eye className="w-4 h-4" />

                                  )}

                                </button>

                                <button

                                  onClick={() => handleOpenEditor(property.id)}

                                  title="Editar"

                                  className="p-1.5 hover:bg-slate-100 rounded-lg text-slate-500 hover:text-blue-600 transition-colors"

                                >

                                  <Edit3 className="w-4 h-4" />

                                </button>

                                <button

                                  onClick={() => handleDeleteProperty(property)}

                                  title="Eliminar"

                                  className="p-1.5 hover:bg-slate-100 rounded-lg text-slate-500 hover:text-red-600 transition-colors"

                                >

                                  <Trash2 className="w-4 h-4" />

                                </button>

                              </div>

                            </td>

                          </tr>

                        ))}

                      </tbody>

                    </table>

                  </div>

                )}

              </div>

            )}

          </div>

        </div>

      </SplitViewLayout>

      {/* Modal de Confirmación Global para Acciones Destructivas/Visibilidad */}

      <ConfirmModal

        isOpen={confirmModal.isOpen}

        title={confirmModal.title}

        message={confirmModal.message}

        variant={confirmModal.variant}

        onConfirm={confirmModal.action}

        onClose={() => setConfirmModal((prev) => ({ ...prev, isOpen: false }))}

      />

    </ModuleLayout>

  );

};

export default AsesorInventoryView;
